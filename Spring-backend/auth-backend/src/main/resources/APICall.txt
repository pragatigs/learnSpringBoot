POST , create user - 
curl -X POST http://localhost:8080/users/create \
  -H "Content-Type: application/json" \
  -d '{"username":"testuser1","password":"test1234"}'

GET ALL - 
curl http://localhost:8080/users/getAll

authenticate db login - 
curl -X POST http://localhost:8080/users/login \
  -H "Content-Type: application/json" \
  -d '{"username": "testuser", "password": "test123"}'

put orders - 
curl -X POST http://localhost:8080/users/putOrders \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer null" \
  -d '{
    "product": "iPhone 15",
    "price": 999.99,
    "orderId": "ORD123456"
  }'


get orders by username - 
curl -X GET http://localhost:8080/users/getOrdersByUser \
  -H "Authorization: Bearer <JWT generated>"


jwt login - 
curl -X POST http://localhost:8080/users/jwtLogin \
  -H "Content-Type: application/json" \
  -d '{"username": "testuser", "password": "test123"}'
